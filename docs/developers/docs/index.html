quickstart_collect.py (Python SDK) pip install gajalogic 
import gajalogic as gj

client = gj.Client(env="sandbox")

# Initiate a UPI collect request against the simulated fabric
txn = client.upi.collect(
    payer_vpa="rahim@okhdfc",
    payee_vpa="bharat-retail@icici",
    amount_inr=249.00,
    idempotency_key="order-88431",
)

print(txn.status)          # SUCCESS
print(txn.pqc.seal)        # ML-DSA-65 (FIPS 204)
print(txn.trace.export())  # architecture-review.json
lookup_vpa.go (Go SDK) go get github.com/gajalogic/gaja-go 
package main

import "github.com/gajalogic/gaja-go"

func main() {
    client := gaja.New(gaja.SandboxEnv)

    // Verify a VPA and confirm it resolves in-memory (no CBS hit)
    lookup, _ := client.UPI.LookupVPA("bharat-retail@icici")

    println(lookup.Resolved)         // true
    println(lookup.Source)           // IN_MEMORY_CACHE
    println(lookup.LatencyMicros)    // 138
}
