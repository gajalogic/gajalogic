Install gajactl Tool

02

Initialize Enclave

Serve Model

Effortless Integration Across Languages

pipeline_inference.py (Python SDK) Python 3.11 
import gajalogic as gj
from transformers import AutoTokenizer

# Connect to local G4 PCIe accelerator ring
device = gj.Device("logic-titan:0", enclave_mode="strict")
print(f"Discovered {device.compute_units} LogicCore units")

# Compile transformer model graph directly into silicon memory
model = gj.Model.from_pretrained(
    "meta-llama/Meta-Llama-3-70B-Instruct",
    device=device,
    precision=gj.Precision.FP8_E4M3,
    max_context_length=8192
)

# Stream tokens with deterministic sub-millisecond inter-token latency
prompt = "Explain quantum error correction using surface codes."
for token in model.generate_stream(prompt, temperature=0.2):
    print(token.text, end="", flush=True)
main.rs (High-Yield Rust Crate) edition 2021 
use gaja_core::prelude::*;
use tokio_stream::StreamExt;

#[tokio::main]
async fn main() -> Result<(), GajaError> {
    // Acquire zero-copy memory ring buffer on PCIe 5.0 slot
    let cluster = Cluster::connect("localhost:8443").await?;
    let session = cluster.session("crypto-enclave-01")?;

    // Dispatch batch requests across 8-card coherent fabric
    let mut stream = session
        .infer_stream(InferenceRequest::new("input_payload.bin"))
        .await?;

    while let Some(chunk) = stream.next().await {
        println!("Received chunk: {:?} in {:?}", chunk.id, chunk.latency);
    }

    Ok(())
}
